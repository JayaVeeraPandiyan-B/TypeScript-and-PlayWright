* Loop - Oru statement ah 1 or more times nama requirement ku etha maari iterate panalam evlo time venum naalum.

* While loop - Oru condition true ah irukura varaikum execute pannum, condition false aana loop stop aagidum.
</> Markdown
```ts
// Print 1 - 10 in Acending order
let i : number = 1;

while (i <= 10) {
    console.log(i);
    i++;
}

// **********
// Even loop Example 1
let i : number = 2;

while (i<=10) {console.log(i); i+=2;}

// **********
// Even loop Example 2
let i : number = 2;

while (i <= 10) {

    if (i % 2 == 0) {console.log(i)} i+=2;
    
}

// **********
// Odd loop Example 1
let i : number = 1;

while(i<=10){console.log(i); i+=2;}

// **********
// Odd loop Example 2
let i : number = 1;

while (i<=10) {
    if(i%2 !=0){console.log(i);} i+=2;
}

// **********
// Print 10 - 1 in decending order
let i : number = 10;

while(i>=1){console.log(i); i--}

// **********
// Print Even from 10 - 2 in decending order example 1
let i : number = 10;

while (i>=2) {
    console.log(i);
    i-=2;
}

// **********
// Print Even from 10 - 2 in decending order example 2
let i : number = 10;

while (i>=2) {
    if (i % 2 == 0) {
        console.log(i);
    }
    i-=2;
}
```

* Do while loop - Loop first oru thadava run aagum, apram condition false aagura varai continue aagum.
</> Markdown
```ts
let num : number = 1;

do {
    console.log(num);
    num++
} while (num<=5); //Acending

// **********

let num : number = 5;

do{console.log(num); num--;} while(num >= 1); //Decending
```

* For loop - Oru condition true ah irukura varaikum loop execute aagum, condition false aana loop stop aagidum.

    For loop work flow = for(initialization(1); condition(2); increment/decrement(4)) {statements (3)}, So order: 1 → 2 → 3 → 4 → 2 → 3 → 4 → 2 → 3 → 4 → 2...
</> Markdown
```ts
// For loop Acending order
for (let num : number = 1; num <= 10; num++) {
    console.log("Now:",num);   
}

// **********
// For loop print even numbers
for (let num:number=2; num <= 10; num++) {if (num % 2 == 0) {console.log(num)}}

// **********
// For loop print odd numbers
for (let num:number=1; num <= 10; num++) {if (num % 2 != 0) {console.log(num)}}

// **********
// For loop Decending order
for (let num:number = 10; num>=1; num--){console.log(num)}

// **********
// For loop print even in decending order
for (let num:number = 10; num>=1; num--){if(num%2==0){console.log(num)}}

// **********
// For loop print odd number in decending order
for (let num:number = 10; num >= 1 ; num--) {
    if (num % 2 != 0) {
        console.log(num);
    }
}
```

* break; - Loop ah immediately stop pannum. (Jumping statement)

    Loop execute aagumbodhu oru specific condition meet aana udane loop ah stop pannum.
</> Markdown
```ts
//Example of break;
for (let num:number = 1; num <= 5; num++)
    {
        if(num >= 5) {
            break;
        }
        console.log(num);
    }
```

* continue; - Current iteration ah skip panni next iteration ku pogum. (Skip statement)

    Oru condition match aana current iteration ah skip panni next iteration ku pogum.
</> Markdown
```ts
// Example of continue; - 1
for (let num:number=1; num <= 10; num++)
{
if(num == 5)
{
continue;
}
console.log(num)
}

// **********
// Example of continue; - 2 (Odd numbers to print 1,3,5,7,9 but expected to print 1 & 9)

for(let num:number=1; num <= 10; num++)
{
if(num % 2 == 0 || num == 3 || num == 5 || num == 7)
{
continue;
}
console.log(num);
}
```
