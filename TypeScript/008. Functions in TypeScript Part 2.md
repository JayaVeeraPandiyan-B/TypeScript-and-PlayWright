* Call Back function: oru function ah arguments ah another function oda parameter ku pass panni later execute panalam athuku peru tha "Call Back".

</> Markdown
```ts
//Example 1 of Call Back

//Function that takes callback function as a parameter
function greet(name:string, msgCall:(msg:string)=>void)  {
    console.log(name);
    console.log("King"); // executing the callback function
}

//CallBack function
function showMessage(msg:string)  {
    console.log(msg);
}

//calling the function by passing the callback function
greet("Jaya", showMessage);

// **********

//Example 2 of Call Back

function resultSum(a:number, b:number, calSum:(result:number)=>void)    {
    let result = a+b;
    calSum(result);
}

function displayResult(result:number):void {
    console.log(result);
}

resultSum(10, 20, displayResult);
```

* Function Overloading: Oru function same name la multiple times create pannalam but with different parameters or return types. Inga 3 different step follow pananum

    * Step 1: Write a signature of function
    * Step 2: Implement a function
    * Step 3: Calling function

 <> Markdown
 ```ts
// **Example 1: Function Overloading - Different parameter types (Data types)**

// Write a signature of function
function getInfo(id:number):string    
function getInfo(name:string):string

// Implement a function
function getInfo(param:number | string) : string { 

    if (typeof param === "number") {
        return (`User ID is ${param}`)
    }

        return (`Username is ${param}`)

}

// Calling function
console.log(getInfo(2000105128));  
console.log(getInfo("Jaya Veera Pandiyan B"));

// **********

// **Example 2: Function Overloading - Different number of parameters**

// Write a signature of function
function add(a:number):number
function add(a:number, b:number):number
function add(a:number, b:number, c:number):number

// Implement a function
function add(a:number, b?:number, c?:number):number {
    if(b !== undefined && c!== undefined)    {
        return a+b+c;
    } if (b !== undefined) {
        return a + b;
    } return a;
}

// Calling function
console.log(add(10)); //Signature 1
console.log(add(10, 10)); //Signature 2
console.log(add(10, 10, 10)); //Signature 3

// **Example 3: Function Overloading - Different return types**

function processInput(input:string):string;
function processInput(input:number):number;

function processInput(input : string | number): string | number
{
    if (typeof input === "string")  {
        return input.toUpperCase();
    } return input * 2;
}

console.log(processInput("jaya veera pandiyan b"));
console.log(processInput(50));

// **Example 4: Function Overloading - Different return types**

function greet(value: string):string;
function greet(value: number):string;
function greet(value: boolean):string;

function greet(value: string | number | boolean): string
{
    if (typeof value === "string")   {
            return(`Hello ${value}`);
    } else if (typeof value === "number")    {
            return(`You're ${value} years old`);
    } let res: string = value? "Married":"Single"
        return res;
}

console.log(greet("Jaya Veera Pandiyan B"));
console.log(greet(34));
console.log(greet(false));
console.log(greet(true));
```
