Example:
</> TypeScript
```ts
//Class
class Student   {
    readonly studentID:number;
    name : string;
    email? : string;
    //schoolName : string;

// Constructor
constructor (id:number, name:string, email?:string)  {
    this.studentID=id;
    this.name=name;
    this.email=email; // email pass panala na ithu undefined
}

//Method
displayInfo() : void    {

    console.log("Student id is:",this.studentID);
    console.log("Student name is:",this.name);

    if(this.email)    {
        console.log("Email id is:", this.email);
    } else  {
        console.log("Oops - Email id not provided!!");
    }
}

}

//Object

let obj1 = new Student(2000105128, "Jaya Veera Pandiyan B");
let obj2 = new Student(200010512, "Jaya Veera Pandiyan B", "jayaveerapandiyanb@gmail.com");

//Execution
console.log("Object 1 with optional email");
obj1.displayInfo();
console.log();
console.log("Object 2 with email");
obj2.displayInfo();

// Output
Object 1 with optional email
Student id is: 2000105128
Student name is: Jaya Veera Pandiyan B   
Oops - Email id not provided!!

Object 2 with email
Student id is: 200010512
Student name is: Jaya Veera Pandiyan B   
Email id is: jayaveerapandiyanb@gmail.com
```
